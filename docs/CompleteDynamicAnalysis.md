### **✅ **Smart Auto-Routing System**
```typescript
// Automatic Floor Assignment Based on Category
const getSmartRouting = (category: string, gender: 'male' | 'female') => {
  const floors = gender === 'male' ? MALE_FLOORS_DATA : FEMALE_FLOORS_DATA;
  
  for (const floor of floors) {
    if (floor.categories.includes(category)) {
      return { 
        floorId: floor.id, 
        floorName: floor.name, 
        attributes: ATTRIBUTE_CONFIGS[category] || null
      };
    }
  }
  // Fallback routing for new categories
};
```
